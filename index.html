Código-Fonte Embutido (C++)


#include <DHT.h>
#define DHTPIN 2
#define DHTTYPE DHT11
DHT dht(DHTPIN, DHTTYPE);

void setup() {
  Serial.begin(9600);
  dht.begin();
  pinMode(8, OUTPUT); // LED Atenção
  pinMode(9, OUTPUT); // LED Crítico
}

void loop() {
  float temp = dht.readTemperature();
  float umid = dht.readHumidity();

  if (isnan(temp)) return;

  if (temp >= 28.0) {
    digitalWrite(9, HIGH); // Alerta Crítico
  } else if (temp >= 25.0) {
    digitalWrite(8, HIGH); // Alerta Atenção
  } else {
    digitalWrite(8, LOW);
    digitalWrite(9, LOW);
  }

  Serial.print(temp); Serial.print(";");
  Serial.print(umid); Serial.println(";");
  delay(2000);
}
